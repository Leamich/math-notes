$$
R_1\circ R_2 \subset A\times C
$$
$$
R_1\subset A\times B,\: R_2\subset B\times C
$$
$$
R_1\circ R_2=\{<a, c>|\exists b\in B,\begin{cases}aR_1b\\bR_2c\end{cases} \}
$$
