1. $\xi \sim\text{Uni}[0, 1]$
$$
\mathbb{E}\xi = \int\limits _{\mathbb{R}}xd\,P_{\xi}(x)=\int\limits_{0}^{1}x\,dx = \frac{x^{2}}{r}\bigg|_{0}^{1}=\frac{1}{2} 
$$
$$
\mathbb{E}\xi^{2} = \int\limits _{\mathbb{R}}x^{2}\, dP_{\xi}(x)=\int\limits_{0}^{1} x^{2}\,dx=\frac{1}{3}
$$

$$
\mathbb{D}\xi = \mathbb{E}\xi^{2} - \mathbb{E}^{2}\xi = \frac{1}{3} - \left( \frac{1}{2} \right)^{2}=\frac{1}{12}
$$

2. $\xi\sim\text{U}[a, b]$. Пусть $\eta\sim\text{Uni}[0, 1]\implies \xi=(b-a)\eta+a\sim\text{Uni}[a, b]$
$$
\mathbb{E}\xi = \mathbb{E}[(b-a)\eta+a] = (b-a)\mathbb{E}\eta + a = \frac{b+a}{2}
$$
$$
\mathbb{D}\xi=\mathbb{D}[(b-a)\eta+a]=(b-a)^{2}\mathbb{D}\eta=\frac{(b-a)^{2}}{12}
$$

3. $\xi\sim \mathcal{N}(0, 1)$, плотность $\frac{1}{\sqrt{2\pi}}e^{ -x^{2}/2 }$
$$
\mathbb{E}\xi=\int\limits _{\mathbb{R}}x\,dP_{\xi}(x)=\int\limits _{\mathbb{R}} x \cdot \frac{1}{\sqrt{2\pi}}\cdot e^{ -x^{2}/2 }\,dx=0\text{ (т.к. нечётная ф-я)}
$$

$$
\begin{gathered}
\mathbb{D}\xi=\mathbb{E}[\xi^{2}] = \int\limits _{\mathbb{R}}x^{2} \cdot \frac{1}{\sqrt{2\pi}}e^{ -x^{2}/2 }\,dx=\left[ \begin{gathered}
xe^{ -x^{2}/2 }=v' \\ v=-e^{ -x^{2}/2 }\\ u=x
\end{gathered} \right] =\\
=\frac{1}{\sqrt{2\pi}}\left( \cancelto{ 0 }{ -xe^{ -x^{2}/2 }\bigg|_{-\infty}^{+\infty} }+\int\limits _{\mathbb{R}}e^{ -x^{2}/2 }\,dx \right) = \frac{1}{\sqrt{2\pi}}\int\limits _{\mathbb{R}} e^{ -x^{2}/2 }\,dx=1
\end{gathered}
$$

4. $\xi\sim \mathcal{N}(a, \sigma^{2})$, возьмём $\eta\sim \mathcal{N}(0, 1)$ и положим $\xi=\sigma \eta+a$, тогда $\xi\sim \mathcal{N}(a, \sigma^{2})$
$$
\mathbb{E}[\xi] = \mathbb{E}[\sigma \eta+a]=\cancelto{ 0 }{ \sigma \mathbb{E}\eta }+a=a
$$
$$
\mathbb{D}[\xi]=\mathbb{D}[\sigma \eta+a]=\sigma^{2}\cancelto{ 1 }{ \mathbb{D}\eta }=\sigma^{2}
$$
